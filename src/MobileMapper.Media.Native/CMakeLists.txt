find_package(FFMPEG REQUIRED)
add_library(MobileMapperMedia SHARED media.cpp)
target_compile_features(MobileMapperMedia PRIVATE cxx_std_20)
target_compile_definitions(MobileMapperMedia PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX MM_EXPORTS)
target_include_directories(MobileMapperMedia PRIVATE ${FFMPEG_INCLUDE_DIRS})
target_link_directories(MobileMapperMedia PRIVATE ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(MobileMapperMedia PRIVATE ${FFMPEG_LIBRARIES} d3d11 dxgi d3dcompiler ole32)
add_executable(native_tests native_tests.cpp)
target_compile_features(native_tests PRIVATE cxx_std_20)
target_link_libraries(native_tests PRIVATE MobileMapperMedia)
add_test(NAME native_tests COMMAND native_tests)
